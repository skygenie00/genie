async function foodAddImgs(x,fl){
 if(!cfg.token){toast('설정에서 토큰을 먼저 넣어줘 — 글만 저장');upsertFood(x);render();return}
 x.imgs=Array.isArray(x.imgs)?x.imgs:[];const st=Date.now().toString(36);let n=0;
 try{
  for(let i=0;i<fl.length;i++){
   toast(`사진 ${i+1}/${fl.length} 압축·업로드…`);
   const b64=await imgCompress(fl[i],1800,0.85);
   const p=`food/img/${x.id}_${st}_${i+1}.jpg`;
   await ghPutRaw(p,b64,`${cfg.person} · food 사진`);x.imgs.push(p);n++;
   if(!x.thumb){const tb=await imgCompress(fl[i],360,0.8);const tp=`food/thumb/${x.id}.jpg`;await ghPutRaw(tp,tb,`${cfg.person} · food 썸네일`);x.thumb=tp;}
  }
 }catch(e){toast('사진 업로드 실패: '+e.message+(n?` — ${n}장만 붙음`:' — 글만 저장'));}
 upsertFood(x);if(n)toast(`사진 ${n}장 붙였어`);render();
